// ETAPA A: chamada genérica ao Gemini (com novas tentativas e troca de modelo)
// Recebe as instruções do sistema, o texto do usuário e a temperatura.
// Devolve o texto da resposta (que esperamos ser um JSON).
//  - Erro temporário (429 por minuto, 500, 503) ou resposta vazia: espera e repete.
//  - Cota diária estourada (429 "PerDay") ou modelo inexistente (404): troca de modelo.
//  - Outros erros (ex.: 400): para na hora e mostra o motivo.
// ============================================================
function chamarGemini(instrucoes, textoUsuario, temperatura) {
  // Lê a chave guardada nas propriedades do script (nunca fica no código)
  const chave = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
  if (!chave) throw new Error("Propriedade GEMINI_API_KEY não encontrada (Parte 2).");

  // Corpo da requisição: instruções do sistema + mensagem + configurações
  const corpo = {
    systemInstruction: { parts: [{ text: instrucoes }] },
    contents: [{ role: "user", parts: [{ text: textoUsuario }] }],
    generationConfig: {
      temperature: temperatura,
      responseMimeType: "application/json" // pede ao modelo que responda em JSON
    }
  };

  const tentativasPorModelo = 2;
  let ultimoErro = "";

  // Percorre os modelos na ordem da lista
  for (let m = 0; m < MODELOS.length; m++) {
    const modelo = MODELOS[m];
    const url = "https://generativelanguage.googleapis.com/v1beta/models/" + modelo + ":generateContent";

    for (let tentativa = 1; tentativa <= tentativasPorModelo; tentativa++) {
      const resposta = UrlFetchApp.fetch(url, {
        method: "post",
        contentType: "application/json",
        headers: { "x-goog-api-key": chave },
        payload: JSON.stringify(corpo),
        muteHttpExceptions: true // não estoura erro: tratamos o código HTTP abaixo
      });
      const codigo = resposta.getResponseCode();
      const texto = resposta.getContentText();

      // ---- Sucesso ----
      if (codigo === 200) {
        const dados = JSON.parse(texto);
        const partes = (dados.candidates && dados.candidates[0] && dados.candidates[0].content &&
                        dados.candidates[0].content.parts) || [];
        // Pega a parte que tem texto (ignora partes internas do modelo)
        const parteTexto = partes.find(function (p) { return typeof p.text === "string" && !p.thought; });

        if (parteTexto && parteTexto.text.trim() !== "") {
          Logger.log("(modelo usado: " + modelo + ")");
          // Remove cercas de markdown (```json) caso o modelo coloque
          return parteTexto.text.replace(/```json|```/g, "").trim();
        }
        ultimoErro = "O Gemini devolveu uma resposta vazia (" + modelo + ").";
        // resposta vazia: tenta de novo (segue o laço)
      } else {
        ultimoErro = "Erro Gemini " + codigo + " (" + modelo + "): " + texto;

        // Modelo inexistente ou cota DIÁRIA estourada: não adianta repetir, vai para o próximo modelo
        if (codigo === 404 || (codigo === 429 && texto.indexOf("PerDay") !== -1)) {
          Logger.log("Modelo " + modelo + " indisponível (" + codigo + "). Tentando o próximo...");
          break;
        }

        // Qualquer erro que não seja temporário (ex.: 400): para tudo e mostra o motivo
        if (codigo !== 429 && codigo !== 500 && codigo !== 503) {
          throw new Error(ultimoErro);
        }
      }

      // Espera antes de repetir no mesmo modelo (5 s)
      if (tentativa < tentativasPorModelo) {
        Logger.log("Tentativa " + tentativa + " falhou. Aguardando para repetir...");
        Utilities.sleep(5000);
      }
    }
  }

  // Nenhum modelo funcionou
  throw new Error(ultimoErro);
}

// ============================================================
// ETAPA B: validar o CNPJ pela conta oficial (módulo 11)
// ============================================================
function cnpjValido(cnpj) {
  // Precisa ter 14 números e não pode ser repetido (ex: 11111111111111)
  if (!cnpj || !/^\d{14}$/.test(cnpj) || /^(\d)\1+$/.test(cnpj)) return false;

  // Calcula um dígito verificador usando os primeiros "tamanho" números
  const calcDigito = function (tamanho) {
    let soma = 0;
    let peso = tamanho - 7;              // pesos começam em 5 (1º dígito) ou 6 (2º dígito)
    for (let i = 0; i < tamanho; i++) {
      soma += Number(cnpj[i]) * peso--;  // multiplica cada número pelo seu peso
      if (peso < 2) peso = 9;            // quando o peso acaba, volta para 9
    }
    const resto = soma % 11;
    return resto < 2 ? 0 : 11 - resto;   // regra oficial do módulo 11
  };

  // Compara os dígitos calculados com os informados (posições 12 e 13)
  return calcDigito(12) === Number(cnpj[12]) && calcDigito(13) === Number(cnpj[13]);
}

// ============================================================
// ETAPA C: consultar a BrasilAPI
// Devolve o objeto da empresa, ou null se o CNPJ não for encontrado.
// ============================================================
function consultarBrasilApi(cnpj) {
  const resposta = UrlFetchApp.fetch("https://brasilapi.com.br/api/cnpj/v1/" + cnpj, {
    muteHttpExceptions: true // 404 não derruba o programa
  });
  if (resposta.getResponseCode() === 404) return null;                 // CNPJ válido, mas inexistente
  if (resposta.getResponseCode() !== 200) {
    throw new Error("Erro BrasilAPI " + resposta.getResponseCode());
  }
  return JSON.parse(resposta.getContentText());
}

// ============================================================
// ETAPA D: preparar dados (enxugar o JSON e calcular meses de atividade)
// A conta de data é feita aqui, em código, pois IAs erram contas de data.
// ============================================================
function prepararDados(d, foco) {
  // Calcula quantos meses a empresa tem de atividade
  const inicio = new Date(d.data_inicio_atividade);   // formato "AAAA-MM-DD"
  const hoje = new Date();
  let meses = null;                                    // fica null se a data não vier
  if (!isNaN(inicio.getTime())) {
    meses = (hoje.getFullYear() - inicio.getFullYear()) * 12 + (hoje.getMonth() - inicio.getMonth());
  }

  // Objeto enxuto: só o que a análise precisa. Coloca "não informado" se o campo faltar
  return {
    foco: foco || [],
    empresa: {
      cnpj: d.cnpj != null ? d.cnpj : "não informado",
      razao_social: d.razao_social || "não informado",
      nome_fantasia: d.nome_fantasia || "não informado",
      situacao_cadastral: d.descricao_situacao_cadastral || "não informado",
      data_inicio_atividade: d.data_inicio_atividade || "não informado",
      meses_de_atividade: meses,
      atividade_principal: d.cnae_fiscal_descricao || "não informado",
      porte: d.porte || d.descricao_porte || "não informado",
      capital_social: d.capital_social != null ? d.capital_social : "não informado",
      municipio: d.municipio || "não informado",
      uf: d.uf || "não informado",
      // Até 5 sócios, só nome e qualificação (economiza tokens)
      socios: (d.qsa || []).slice(0, 5).map(function (s) {
        return { nome: s.nome_socio, qualificacao: s.qualificacao_socio };
      })
    }
  };
}

// ORQUESTRADOR: executa o fluxo completo para uma mensagem
// Sempre devolve um objeto com o campo "sucesso".
// ============================================================
function processar(mensagem) {
  // 1) Gemini #1 interpreta a mensagem
  let interpretado;
  try {
    interpretado = JSON.parse(chamarGemini(PROMPT_INTERPRETADOR, mensagem, 0));
  } catch (e) {
    // Registra o motivo real no log antes de responder ao usuário
    Logger.log("ERRO no Gemini #1 (interpretador): " + e.message);
    return { sucesso: false, mensagem: "Não consegui interpretar sua mensagem. Tente reescrever." };
  }

  // 2) Regras de decisão (validação da entrada)
  if (interpretado.intent !== "consultar_cnpj") {
    return { sucesso: false, mensagem: "Só consigo ajudar com consultas de CNPJ de empresas." };
  }
  if (!interpretado.cnpj) {
    return { sucesso: false, mensagem: "Informe o CNPJ que deseja consultar." };
  }
  if (!cnpjValido(interpretado.cnpj)) {
    return { sucesso: false, mensagem: "O CNPJ informado é inválido. Confira os números." };
  }

  // 3) Consulta a BrasilAPI
  const empresa = consultarBrasilApi(interpretado.cnpj);
  if (!empresa) {
    return { sucesso: false, mensagem: "CNPJ válido, mas não encontrado na base da Receita Federal." };
  }

  // 4) Prepara os dados (enxuga e calcula meses de atividade)
  const dados = prepararDados(empresa, interpretado.foco);

  // 5) Gemini #2 analisa o risco (temperatura baixa, mas não zero)
  const textoAnalise = "Foco da análise: " + JSON.stringify(dados.foco) +
                       "\nDados da empresa: " + JSON.stringify(dados.empresa);
  let analise;
  try {
    analise = JSON.parse(chamarGemini(PROMPT_ANALISTA, textoAnalise, 0.2));
  } catch (e) {
    // Registra o motivo real no log antes de responder ao usuário
    Logger.log("ERRO no Gemini #2 (analista): " + e.message);
    return { sucesso: false, mensagem: "Falha ao gerar a análise. Tente novamente." };
  }

  // 6) Corrige em código o que não deve depender da IA
  if (dados.empresa.meses_de_atividade !== null) {
    analise.tempo_de_atividade_anos = Math.round((dados.empresa.meses_de_atividade / 12) * 10) / 10;
  }

  return Object.assign({ sucesso: true }, analise);
}

// ============================================================
// ETAPA E: registrar a consulta no Google Sheets (integração Workspace)
// Cada consulta vira uma linha. Se algo falhar aqui, o usuário NÃO é afetado:
// o erro só vai para o log.
// Requer a propriedade do script SHEET_ID (ID da planilha).
// ============================================================
function registrarConsulta(mensagem, r) {
  try {
    const id = PropertiesService.getScriptProperties().getProperty("SHEET_ID");
    if (!id) {
      Logger.log("SHEET_ID não configurado: consulta não registrada.");
      return;
    }

    // Usa a primeira aba da planilha e garante o nome "Consultas"
    const planilha = SpreadsheetApp.openById(id);
    const aba = planilha.getSheets()[0];
    if (aba.getName() !== "Consultas") aba.setName("Consultas");

    // Se a aba estiver vazia, cria o cabeçalho
    if (aba.getLastRow() === 0) {
      aba.appendRow(["Data/Hora", "Mensagem", "CNPJ", "Empresa", "Situação", "Risco", "Resultado", "Resposta"]);
      aba.setFrozenRows(1);
      aba.getRange(1, 1, 1, 8).setFontWeight("bold");
    }

    // Monta a linha. Em caso de erro de entrada, a empresa vem vazia ("-")
    const emp = r.empresa || {};
    const linha = [
      Utilities.formatDate(new Date(), "America/Sao_Paulo", "dd/MM/yyyy HH:mm:ss"),
      mensagem,
      emp.cnpj || "-",
      emp.razao_social || "-",
      r.situacao_cadastral || "-",
      r.nivel_risco || "-",
      r.sucesso ? "sucesso" : "sem sucesso",
      r.sucesso ? r.resposta_formatada : r.mensagem
    ];

    // Formata a coluna do CNPJ como TEXTO antes de gravar, senão o Sheets
    // apaga os zeros à esquerda (00000000000191 viraria 191)
    const numeroLinha = aba.getLastRow() + 1;
    aba.getRange(numeroLinha, 3).setNumberFormat("@");
    aba.getRange(numeroLinha, 1, 1, linha.length).setValues([linha]);
  } catch (e) {
    Logger.log("ERRO ao registrar no Sheets: " + e.message);
  }
}

// ============================================================
// WEBHOOK (POST): recebe {"mensagem": "..."} e devolve JSON
// ============================================================
function doPost(e) {
  let resultado;
  try {
    const corpo = JSON.parse(e.postData.contents);
    const mensagem = String(corpo.mensagem || "");
    resultado = processar(mensagem);
    registrarConsulta(mensagem, resultado); // grava a consulta na planilha
  } catch (erro) {
    resultado = { sucesso: false, mensagem: "Requisição inválida. Envie JSON com o campo 'mensagem'." };
  }
  return ContentService.createTextOutput(JSON.stringify(resultado))
    .setMimeType(ContentService.MimeType.JSON);
}

// WEBHOOK (GET): permite testar no navegador com ?mensagem=...
function doGet(e) {
  const mensagem = (e && e.parameter && e.parameter.mensagem) || "";
  let resultado;
  if (mensagem) {
    resultado = processar(mensagem);
    registrarConsulta(mensagem, resultado); // grava a consulta na planilha
  } else {
    resultado = { sucesso: false, mensagem: "Use ?mensagem=... na URL." };
  }
  return ContentService.createTextOutput(JSON.stringify(resultado))
    .setMimeType(ContentService.MimeType.JSON);
}
